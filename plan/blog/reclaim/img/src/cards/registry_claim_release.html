1pub(crate) fn claim(&self) -> &Record<P> {
2    for rec in self.iter() {                       // ③ iter loads head with Acquire
3        if rec.in_use
4            .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)   // ④ one winner
5            .is_ok()
6        {
7            return rec;                            // the loser moves on to the next record
8        }
9    }
10    self.push_new()                                // nothing free: a new record
11}

13/// # Safety: the caller holds the claim, and does not use the record again until it claims it again.
14pub(crate) unsafe fn release(&self) {
15    self.payload.clear();                          // first: the payload means "using nothing"
16    self.in_use.store(false, Ordering::Release);   // ② then free; pairs with claim's Acquire
17}
