1pub(crate) unsafe fn push<T: Retirable>(&self, obj: *mut T) {
2    let link: &RetireLink = unsafe { (*obj).retire_link() };
3    let link_ptr = link as *const RetireLink as *mut RetireLink;
4
5    let in_a_pile = link.reclaim.with(|p| unsafe { (*p).is_some() });
6    assert!(!in_a_pile, "object retired twice");
7
8    link.obj.with_mut(|p| unsafe { *p = obj.cast() });                 // ① written before
9    link.reclaim.with_mut(|p| unsafe { *p = Some(reclaim_one::<T>) }); // ① the link is visible
10
11    let mut top = self.head.load(Ordering::Relaxed);
12    loop {
13        link.next.store(top, Ordering::Relaxed);                       // ① still private
14        match self.head.compare_exchange_weak(
15            top, link_ptr, Ordering::Release, Ordering::Relaxed,       // ② publish
16        ) {
17            Ok(_) => return,
18            Err(current) => top = current,
19        }
20    }
21}
