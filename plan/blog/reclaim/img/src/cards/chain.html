retire.rs — a chain built privately, attached with one exchange
1pub(crate) struct Chain {
2    top: *mut RetireLink,         // newest link
3    bottom: *mut RetireLink,      // oldest link: the only one that points into the pile
4}
5
6impl Chain {
7    pub(crate) unsafe fn add<T: Retirable>(&mut self, obj: *mut T) {
8        let link: &RetireLink = unsafe { (*obj).retire_link() };
9        let link_ptr = link as *const RetireLink as *mut RetireLink;
10        // ... the same double-retire check and obj / reclaim writes as push ...
11        if self.top.is_null() {
12            self.bottom = link_ptr;                         // the first link stays the bottom
13        } else {
14            link.next.store(self.top, Ordering::Relaxed);   // private: Relaxed is enough
15        }
16        self.top = link_ptr;
17    }
18}
19
20impl RetiredList {
21    pub(crate) fn push_chain(&self, chain: Chain) {
22        if chain.top.is_null() { return; }
23        let mut top = self.head.load(Ordering::Relaxed);
24        loop {
25            // ① a retry changes only the bottom's `next`, to the current `head`; the links above never change
26            unsafe { (*chain.bottom).next.store(top, Ordering::Relaxed) };
27            match self.head.compare_exchange_weak(
28                top, chain.top, Ordering::Release, Ordering::Relaxed,   // ② publishes the whole chain
29            ) {
30                Ok(_) => return,
31                Err(current) => top = current,
32            }
33        }
34    }
35}
