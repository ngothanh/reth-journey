retire.rs — take everything at once, then free link by link
1pub(crate) fn take_all(&self) -> *mut RetireLink {
2    self.head.swap(null_mut(), Ordering::Acquire)       // ③ own the whole chain, then open it
3}
4
5pub(crate) unsafe fn free_all(mut link: *mut RetireLink) {
6    while !link.is_null() {
7        let next = unsafe { (*link).next.load(Ordering::Relaxed) };   // ④ BEFORE reclaim
8        let obj = unsafe { (*link).obj.with(|p| *p) };
9        let reclaim = unsafe { (*link).reclaim.with(|p| *p).unwrap() };
10
11        // give the link back empty, so a pooled object can be retired again;
12        // must happen before reclaim: after it the link is freed memory
13        unsafe { (*link).obj.with_mut(|p| *p = null_mut()) };
14        unsafe { (*link).reclaim.with_mut(|p| *p = None) };
15
16        unsafe { reclaim(obj) };                         // the link was inside obj: gone now
17        link = next;
18    }
19}
