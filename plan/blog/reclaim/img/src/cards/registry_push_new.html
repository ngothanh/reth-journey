1pub(crate) fn push_new(&self) -> &Record<P> {
2    // once, before the loop: a retry must still have the record to write into
3    let new_record = Box::into_raw(Box::new(CachePadded::new(Record::<P>::new())));
4    let mut top = self.head.load(Ordering::Relaxed);
5    loop {
6        // SAFETY: not in the list yet, so no other thread can reach it
7        unsafe { (&*new_record).next.with_mut(|p| *p = top) };
8        match self.head.compare_exchange_weak(
9            top, new_record,
10            Ordering::Release,                // ① publishes the record, fully built
11            Ordering::Relaxed,
12        ) {
13            Ok(_) => return unsafe { &*new_record },   // SAFETY: records are never freed (Rule 1)
14            Err(current) => top = current,             // retry on the head that won
15        }
16    }
17}
