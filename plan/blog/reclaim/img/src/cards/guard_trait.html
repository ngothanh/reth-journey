1pub trait Guard {
2    /// Announce `addr`, then check that `root` still holds it.
3    /// Ok(addr):     `addr` is protected and may be opened.
4    /// Err(current): the root moved on; do not open `addr`, retry with `current`.
5    fn try_protect<T>(&mut self, addr: *mut T, root: &Root<T>)   // ① &mut self: reuse one guard
6        -> Result<*mut T, *mut T>;                                // ② a raw pointer, not &'g T
7
8    /// Provided: the retry loop, written once, for callers with nothing better to do.
9    fn protect<T>(&mut self, root: &Root<T>) -> *mut T {
10        let mut addr = root.ptr().load(Ordering::Acquire);
11        loop {
12            match self.try_protect(addr, root) {
13                Ok(p) => return p,
14                Err(current) => addr = current,
15            }
16        }
17    }
18}
