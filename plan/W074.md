    pub struct RaftBackedEngine {
        engine: Engine,                        // local OrderBook + perps state from W67
        raft: RaftNode<MatchCommand, MatchReply>,
        applied_index: u64,
        deterministic_clock: SequenceClock,    // monotonic tick driven by raft index, NOT wall clock

    pub enum MatchCommand {
        NewOrder { id: OrderId, order: Order, submitter_ts: SeqTime },
        Cancel { id: OrderId },
        ApplyFunding { symbol: Symbol, rate_bps: i64 },
        LiquidationStep { account: AccountId, qty: u64, mark: Price },
        ADLStep { symbol: Symbol, victims: Vec<ADLEntry> },
        AdvanceClock { to_seq: u64 },           // explicit clock-tick command
    }
    impl RaftBackedEngine {
        // Leader path
        pub async fn submit(&mut self, cmd: MatchCommand) -> Result<CommandIndex> {
            let proposed = self.raft.propose(cmd).await?;
            Ok(proposed.index)
        }
        // Replica path — invoked by raft on every replica
        fn apply(&mut self, index: u64, cmd: MatchCommand) -> MatchReply {
            assert_eq!(index, self.applied_index + 1, "non-contiguous apply");
            let reply = match cmd { /* dispatch to Engine */ };
            self.applied_index = index;
            reply
        }

  - **Design**:
    ```rust
    pub struct NormalReplica {
        pub id: ReplicaId,
        pub view: View,
        pub status: ReplicaStatus,            // == Normal
        pub op_num: u64,                      // highest op assigned (leader) or seen (backup)
        pub commit_num: u64,                  // highest op committed
        pub log: Log,                          // op_num → Op, persisted in wal
        pub client_table: AHashMap<ClientId, ClientRequest>,  // dedupe + last-reply cache
        pub prepare_ok: AHashMap<u64, BitSet>,  // op_num → set of replicas acking
        pub state_machine: Box<dyn StateMachine>,
    }
    pub struct Prepare {
        pub view_num: ViewNum,
        pub op_num: u64,
        pub op: Op,                            // client request body
        pub commit_num: u64,                   // leader's known commit
    }
    pub struct PrepareOk {
        pub view_num: ViewNum,
        pub op_num: u64,
        pub replica_id: ReplicaId,
    }
    impl NormalReplica {
        // Leader path
        pub fn handle_client_request(&mut self, req: ClientRequest) -> Result<()> {
            assert!(self.view.is_leader(self.id));
            // Dedupe via client_table
            if let Some(prev) = self.client_table.get(&req.client) {
                if prev.req_num >= req.req_num { return Err(/*duplicate*/); }
            }
            self.op_num += 1;
            self.log.append(self.op_num, req.op.clone())?;  // wal-durable
            self.broadcast(Prepare { view_num: self.view.view_num, op_num: self.op_num, op: req.op, commit_num: self.commit_num });
            Ok(())
        }
        // Backup path
        pub fn on_prepare(&mut self, p: Prepare) -> Result<PrepareOk> {
            if p.view_num != self.view.view_num { return Err(/*view-mismatch*/); }
            if p.op_num != self.op_num + 1 { /* gap: request state transfer */ return Err(/*gap*/); }
            self.log.append(p.op_num, p.op)?;
            self.op_num = p.op_num;
            self.advance_commit(p.commit_num);
            Ok(PrepareOk { view_num: self.view.view_num, op_num: p.op_num, replica_id: self.id })
        }
        // Leader: collect acks, commit at quorum (f+1 including self)
        pub fn on_prepare_ok(&mut self, ack: PrepareOk) {
            assert!(self.view.is_leader(self.id));
            if ack.view_num != self.view.view_num { return; }
            let acks = self.prepare_ok.entry(ack.op_num).or_default();
            acks.insert(ack.replica_id as usize);
            acks.insert(self.id as usize);  // count self
            if acks.count_ones() as u8 >= self.view.quorum() {
                self.advance_commit(ack.op_num);
            }
        }
        fn advance_commit(&mut self, up_to: u64) {
            while self.commit_num < up_to {
                self.commit_num += 1;
                let op = self.log.get(self.commit_num);
                let reply = self.state_machine.apply(op);
                /* leader: ship reply to client; backup: silently apply */
            }
        }
    }
    ```
    **Invariants**: (i) `op_num` is gapless — every replica's log[1..=op_num] is contiguous. (ii) `commit_num <= op_num` always. (iii) Leader assigns op_num monotonically; backups only accept `op_num = self.op_num + 1` (no skipping ahead — triggers state transfer instead). (iv) Prepare/PrepareOk processed only when `status == Normal` AND `view_num == self.view.view_num`. (v) `state_machine.apply` called exactly once per op_num across the cluster — replicas check via commit_num.
